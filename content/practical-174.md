# Check Pass or Fail for 5 Subjects

sub1 = int(input("Enter marks for Subject 1: "))
sub2 = int(input("Enter marks for Subject 2: "))
sub3 = int(input("Enter marks for Subject 3: "))
sub4 = int(input("Enter marks for Subject 4: "))
sub5 = int(input("Enter marks for Subject 5: "))

if sub1 >= 35 and sub2 >= 35 and sub3 >= 35 and sub4 >= 35 and sub5 >= 35:
    print("Result: PASS")
else:
    print("Result: FAIL")

Output:
Enter marks for Subject 1: 65
Enter marks for Subject 2: 72
Enter marks for Subject 3: 48
Enter marks for Subject 4: 55
Enter marks for Subject 5: 61

Result: PASS
